[PCOMM SCRIPT HEADER]
LANGUAGE=VBSCRIPT
DESCRIPTION=Вставка текста по выбору причины (меню)
[PCOMM SCRIPT SOURCE]
OPTION EXPLICIT
autECLSession.SetConnectionByName(ThisSessionName)

' ================== НАСТРОЙКИ ==================
Const MENU_HTA    = "C:\EQ\EqMenu.hta"
Const CLEAR_FIELD = True     ' очищать поле от курсора до конца перед вставкой

' "Группа|Название в меню|Текст для вставки|ENTER (необязательно)"
' Подстановки: {DATE} - сегодня, {ASK} - ввести текст вручную в окне
Dim ACTIONS
ACTIONS = Array( _
    "Бронь|Бронь|БРОНЬ ДЕНЕЖНЫХ СРЕДСТВ", _
    "Бронь|Снятие брони|СНЯТИЕ БРОНИ ОТ {DATE}", _
    "Отзыв|По заявлению клиента|ОТЗЫВ ПО ЗАЯВЛЕНИЮ КЛИЕНТА", _
    "Отзыв|Ошибка в реквизитах|ОТЗЫВ: ОШИБКА В РЕКВИЗИТАХ", _
    "Отзыв|Недостаточно средств|ОТЗЫВ: НЕДОСТАТОЧНО СРЕДСТВ", _
    "Отзыв|Другая причина|ОТЗЫВ: {ASK}", _
    "Прочее|Свой текст|{ASK}" _
)
' ===============================================

Dim ps, fso, sh
Set ps  = autECLSession.autECLPS
Set fso = CreateObject("Scripting.FileSystemObject")
Set sh  = CreateObject("WScript.Shell")

subMain

Sub subMain()
    Dim row, col, maxLen, tmp, inF, outF, f, i, p, idx, extra, text

    row = ps.CursorPosRow
    col = ps.CursorPosCol

    maxLen = FieldRoomAtCursor(row, col)
    If maxLen = 0 Then
        MsgBox "Курсор не в поле ввода. Поставьте курсор в нужное поле и повторите.", vbExclamation, "Вставка"
        Exit Sub
    End If
    If Not fso.FileExists(MENU_HTA) Then
        MsgBox "Не найден файл меню: " & MENU_HTA, vbCritical, "Вставка"
        Exit Sub
    End If

    ' --- передать пункты в меню ---
    tmp  = sh.ExpandEnvironmentStrings("%TEMP%")
    inF  = tmp & "\eq_menu_in.txt"
    outF = tmp & "\eq_menu_out.txt"
    If fso.FileExists(outF) Then fso.DeleteFile outF

    Set f = fso.CreateTextFile(inF, True, True)          ' Unicode
    f.WriteLine CStr(maxLen)
    For i = 0 To UBound(ACTIONS)
        p = Split(ACTIONS(i) & "|||", "|")
        f.WriteLine p(0) & "|" & p(1) & "|" & p(2)
    Next
    f.Close

    ' --- показать меню и дождаться выбора ---
    sh.Run "mshta.exe """ & MENU_HTA & """", 1, True
    If Not fso.FileExists(outF) Then Exit Sub            ' нажали Esc / закрыли окно

    Set f = fso.OpenTextFile(outF, 1, False, -1)
    idx = CInt(Trim(f.ReadLine))
    extra = ""
    If Not f.AtEndOfStream Then extra = f.ReadAll
    f.Close
    fso.DeleteFile outF

    ' --- собрать текст ---
    p = Split(ACTIONS(idx) & "|||", "|")
    text = Replace(p(2), "{ASK}", extra)
    text = Replace(text, "{DATE}", CStr(Date))

    If maxLen > 0 And Len(text) > maxLen Then
        If MsgBox("Текст длиннее поля (" & Len(text) & " > " & maxLen & " символов)." & vbCrLf & _
                  "Обрезать до:" & vbCrLf & Left(text, maxLen) & " ?", _
                  vbYesNo + vbQuestion, "Вставка") = vbNo Then Exit Sub
        text = Left(text, maxLen)
    End If

    ' --- вставить ---
    ps.SetCursorPos row, col
    If CLEAR_FIELD Then ps.SendKeys "[eraseeof]"
    ps.SetText text, row, col
    If UCase(Trim(p(3))) = "ENTER" Then ps.SendKeys "[enter]"
End Sub

' Сколько символов помещается от курсора до конца поля.
' 0 - курсор не в поле ввода, -1 - определить не удалось
Function FieldRoomAtCursor(row, col)
    Dim fl, f
    FieldRoomAtCursor = -1
    On Error Resume Next
    Set fl = ps.autECLFieldList
    fl.Refresh
    Set f = fl.FindFieldByRowCol(row, col)
    If Err.Number <> 0 Or f Is Nothing Then Err.Clear: Exit Function
    If f.Protected Then FieldRoomAtCursor = 0: Exit Function
    FieldRoomAtCursor = f.Length - ((row - f.StartRow) * ps.NumCols + (col - f.StartCol))
    On Error GoTo 0
End Function
