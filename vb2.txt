[PCOMM SCRIPT HEADER]
LANGUAGE=VBSCRIPT
DESCRIPTION=Вставка текста по выбору причины
[PCOMM SCRIPT SOURCE]
OPTION EXPLICIT
autECLSession.SetConnectionByName(ThisSessionName)

' ================== НАСТРОЙКИ ==================
' "Название в меню|Текст для вставки|ENTER (необязательно)"
' Подстановка: {DATE} - сегодня
Dim ACTIONS
ACTIONS = Array( _
    "Бронь|БРОНЬ ДЕНЕЖНЫХ СРЕДСТВ", _
    "Снятие брони|СНЯТИЕ БРОНИ ОТ {DATE}", _
    "Отзыв - по заявлению клиента|ОТЗЫВ ПО ЗАЯВЛЕНИЮ КЛИЕНТА", _
    "Отзыв - ошибка в реквизитах|ОТЗЫВ: ОШИБКА В РЕКВИЗИТАХ ПОЛУЧАТЕЛЯ", _
    "Отзыв - недостаточно средств|ОТЗЫВ: НЕДОСТАТОЧНО СРЕДСТВ НА СЧЕТЕ ПЛАТЕЛЬЩИКА" _
)

Const CLEAR_FIELD = True     ' стирать старый текст в поле (дополнять пробелами до конца поля)
Const SAME_COLUMN = True     ' переносить только в поля ниже, начинающиеся в той же колонке
Const MAX_FIELDS  = 5        ' максимум полей для переноса (включая первое)
Const CLEAR_REST  = False    ' True - очищать оставшиеся поля, если текст закончился раньше
' ===============================================

Dim ps
Set ps = autECLSession.autECLPS

subMain

Sub subMain()
    Dim row, col, menu, i, p, choice, text
    Dim fields, cur, pos, s, fRow, fCol, fLen, baseCol, lastRow
    Dim tRow(), tCol(), tLen(), n, rest, chunk

    row = ps.CursorPosRow
    col = ps.CursorPosCol

    ' --- меню ---
    menu = "Выберите причину:" & vbCrLf & vbCrLf
    For i = 0 To UBound(ACTIONS)
        p = Split(ACTIONS(i), "|")
        menu = menu & "  " & (i + 1) & " - " & p(0) & vbCrLf
    Next
    menu = menu & vbCrLf & "Введите номер:"

    choice = InputBox(menu, "Вставка в Equation", "1")
    If choice = "" Then Exit Sub
    If Not IsNumeric(choice) Then Exit Sub
    choice = CInt(choice) - 1
    If choice < 0 Or choice > UBound(ACTIONS) Then Exit Sub

    p = Split(ACTIONS(choice) & "||", "|")
    text = Replace(p(1), "{DATE}", CStr(Date))

    ' --- поля ввода на экране ---
    fields = InputFields()
    If UBound(fields) < 0 Then
        ' список полей недоступен - печатаем с позиции курсора
        TypeAt row, col, text
    Else
        ' поле, в котором стоит курсор
        cur = -1
        pos = (row - 1) * ps.NumCols + col
        For i = 0 To UBound(fields)
            s = Split(fields(i), "|")
            fRow = CInt(s(0)): fCol = CInt(s(1)): fLen = CInt(s(2))
            If pos >= (fRow - 1) * ps.NumCols + fCol And pos <= (fRow - 1) * ps.NumCols + fCol + fLen - 1 Then
                cur = i: Exit For
            End If
        Next
        If cur = -1 Then
            MsgBox "Курсор не в поле ввода. Поставьте курсор в нужное поле и повторите.", vbExclamation, "Вставка"
            Exit Sub
        End If

        ' поля для заполнения: первое - от курсора, дальше - следующие поля
        ReDim tRow(MAX_FIELDS - 1): ReDim tCol(MAX_FIELDS - 1): ReDim tLen(MAX_FIELDS - 1)
        tRow(0) = row
        tCol(0) = col
        tLen(0) = (fRow - 1) * ps.NumCols + fCol + fLen - pos
        baseCol = fCol
        lastRow = fRow
        n = 1
        For i = cur + 1 To UBound(fields)
            If n >= MAX_FIELDS Then Exit For
            s = Split(fields(i), "|")
            If Not SAME_COLUMN Or (CInt(s(1)) = baseCol And CInt(s(0)) > lastRow) Then
                tRow(n) = CInt(s(0)): tCol(n) = CInt(s(1)): tLen(n) = CInt(s(2))
                lastRow = tRow(n)
                n = n + 1
            End If
        Next

        ' раскладываем текст по полям: каждое поле заполняется до конца
        rest = text
        For i = 0 To n - 1
            If rest = "" Then
                If CLEAR_REST Then PutText tRow(i), tCol(i), "", tLen(i) Else Exit For
            Else
                chunk = NextChunk(rest, tLen(i))
                PutText tRow(i), tCol(i), chunk, tLen(i)
            End If
        Next
    End If

    If UCase(Trim(p(2))) = "ENTER" Then ps.SendKeys "[enter]"
End Sub

' Отрезать от rest ровно w символов (поле заполняется до конца, слова не учитываются)
Function NextChunk(rest, w)
    NextChunk = Left(rest, w)
    rest = Mid(rest, w + 1)
End Function

' Быстрая запись в поле: текст дополняется пробелами до конца поля,
' поэтому старое содержимое стирается без нажатий клавиш и без ожиданий
Sub PutText(r, c, s, w)
    If CLEAR_FIELD Then
        ps.SetText Left(s & Space(w), w), r, c
    ElseIf s <> "" Then
        ps.SetText s, r, c
    End If
End Sub

' Запасной вариант, если список полей недоступен
Sub TypeAt(r, c, s)
    ps.SetCursorPos r, c
    If CLEAR_FIELD Then ps.SendKeys "[eraseeof]"
    If s <> "" Then ps.SendKeys Replace(Replace(s, "[", "[["), "]", "]]")
End Sub

' Все поля ввода экрана: "строка|колонка|длина"
Function InputFields()
    Dim fl, i, f, arr(), n
    InputFields = Array()
    n = -1
    On Error Resume Next
    Set fl = ps.autECLFieldList
    fl.Refresh
    If Err.Number <> 0 Then Err.Clear: Exit Function
    ReDim arr(fl.Count)
    For i = 1 To fl.Count
        Set f = fl(i)
        If Err.Number = 0 Then
            If Not f.Protected Then
                n = n + 1
                arr(n) = f.StartRow & "|" & f.StartCol & "|" & f.Length
            End If
        End If
        Err.Clear
    Next
    On Error GoTo 0
    If n >= 0 Then
        ReDim Preserve arr(n)
        InputFields = arr
    End If
End Function
