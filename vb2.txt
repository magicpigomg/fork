[PCOMM SCRIPT HEADER]
LANGUAGE=VBSCRIPT
DESCRIPTION=Вставка текста по выбору причины
[PCOMM SCRIPT SOURCE]
OPTION EXPLICIT
autECLSession.SetConnectionByName(ThisSessionName)

' ================== НАСТРОЙКИ ==================
' "Название в меню|Текст для вставки|ENTER (необязательно)"
' Подстановки: {DATE} - сегодня, {ASK} - спросить текст вручную
Dim ACTIONS
ACTIONS = Array( _
    "Бронь|БРОНЬ ДЕНЕЖНЫХ СРЕДСТВ", _
    "Снятие брони|СНЯТИЕ БРОНИ ОТ {DATE}", _
    "Отзыв - по заявлению клиента|ОТЗЫВ ПО ЗАЯВЛЕНИЮ КЛИЕНТА", _
    "Отзыв - ошибка в реквизитах|ОТЗЫВ: ОШИБКА В РЕКВИЗИТАХ", _
    "Отзыв - отсутствие средств|ОТЗЫВ: НЕДОСТАТОЧНО СРЕДСТВ", _
    "Отзыв - другая причина|ОТЗЫВ: {ASK}", _
    "Свой текст|{ASK}" _
)
Const CLEAR_FIELD = True     ' очищать поле от курсора до конца перед вставкой
' ===============================================

Dim ps
Set ps = autECLSession.autECLPS

subMain

Sub subMain()
    Dim menu, i, p, choice, text, row, col, maxLen, extra

    autECLSession.autECLOIA.WaitForInputReady 10000
    row = ps.CursorPosRow
    col = ps.CursorPosCol

    ' --- проверить, что курсор в поле ввода ---
    maxLen = FieldRoomAtCursor(row, col)
    If maxLen = 0 Then
        MsgBox "Курсор не в поле ввода. Поставьте курсор в нужное поле и повторите.", vbExclamation, "Вставка"
        Exit Sub
    End If

    ' --- меню ---
    menu = "Выберите причину:" & vbCrLf & vbCrLf
    For i = 0 To UBound(ACTIONS)
        p = Split(ACTIONS(i), "|")
        menu = menu & "  " & (i + 1) & " - " & p(0) & vbCrLf
    Next
    menu = menu & vbCrLf & "Введите номер:"

    choice = InputBox(menu, "Вставка в Equation", "1")
    If choice = "" Then Exit Sub                                   ' Отмена
    If Not IsNumeric(choice) Then MsgBox "Нужно ввести номер", vbExclamation, "Вставка": Exit Sub
    choice = CInt(choice) - 1
    If choice < 0 Or choice > UBound(ACTIONS) Then MsgBox "Нет такого пункта", vbExclamation, "Вставка": Exit Sub

    p = Split(ACTIONS(choice), "|")

    ' --- собрать текст ---
    text = p(1)
    If InStr(text, "{ASK}") > 0 Then
        extra = InputBox("Введите текст:", p(0), "")
        If extra = "" Then Exit Sub
        text = Replace(text, "{ASK}", extra)
    End If
    text = Replace(text, "{DATE}", CStr(Date))

    ' --- длина поля ---
    If maxLen > 0 And Len(text) > maxLen Then
        If MsgBox("Текст длиннее поля (" & Len(text) & " > " & maxLen & " символов)." & vbCrLf & _
                  "Обрезать до:" & vbCrLf & Left(text, maxLen) & " ?", _
                  vbYesNo + vbQuestion, "Вставка") = vbNo Then Exit Sub
        text = Left(text, maxLen)
    End If

    ' --- вставить ---
    ps.SetCursorPos row, col
    If CLEAR_FIELD Then ps.SendKeys "[eraseeof]"
    ps.SetText text, row, col

    If UBound(p) >= 2 Then
        If UCase(Trim(p(2))) = "ENTER" Then ps.SendKeys "[enter]"
    End If
End Sub

' Сколько символов помещается от курсора до конца поля.
' 0 - курсор не в поле ввода, -1 - определить не удалось (вставляем без проверки)
Function FieldRoomAtCursor(row, col)
    Dim fl, f
    FieldRoomAtCursor = -1
    On Error Resume Next
    Set fl = ps.autECLFieldList
    fl.Refresh
    Set f = fl.FindFieldByRowCol(row, col)
    If Err.Number <> 0 Or f Is Nothing Then Err.Clear: Exit Function
    If f.Protected Then FieldRoomAtCursor = 0: Exit Function
    FieldRoomAtCursor = f.Length - ((row - f.StartRow) * ps.NumCols + (col - f.StartCol))
    On Error GoTo 0
End Function
