[PCOMM SCRIPT HEADER]
LANGUAGE=VBSCRIPT
DESCRIPTION=Equation: сбор полей документа по страницам в файл
[PCOMM SCRIPT SOURCE]
OPTION EXPLICIT
autECLSession.SetConnectionByName(ThisSessionName)

' ================== НАСТРОЙКИ ==================
Const OUT_DIR      = "C:\EQ\"
Const SEP          = ";"
Const UNICODE_FILE = False        ' True - Unicode, False - ANSI (1251)
Const MSG_ROW      = 24           ' строка сообщений Equation
Const TIMEOUT_MS   = 10000
Const SHOW_MSGBOX  = True

' ================== СЦЕНАРИЙ ==================
' Команды (разделитель "|"):
'   READ|Название|строка|колонка|длина     - прочитать поле по координатам
'   LABEL|Название|Текст подписи|длина     - прочитать значение после подписи
'   PGDN                                   - Page Down
'   PGUP                                   - Page Up
'   F2 / F12 / F3 / ENTER                  - нажать клавишу
'   KEY|[pf5]                              - любая клавиша в формате PCOMM
'   WAIT|Текст                             - ждать появления текста на экране
'   PAUSE|мс                               - просто пауза
Dim STEPS
STEPS = Array( _
    "READ|Номер документа|4|20|16", _
    "READ|Дата|5|20|10", _
    "READ|Сумма|10|20|18", _
    "PGDN", _
    "READ|Счёт дебета|7|20|20", _
    "READ|Счёт кредита|8|20|20", _
    "PGDN", _
    "LABEL|Назначение|Назначение:|60", _
    "F2", _
    "READ|Доп. реквизит|6|25|30", _
    "F12" _
)
' ===============================================

Dim ps, oia, headers, values, errText
Set ps  = autECLSession.autECLPS
Set oia = autECLSession.autECLOIA

subMain

Sub subMain()
    Dim i, p, cmd
    headers = Array("Дата выгрузки")
    values  = Array(CStr(Now()))
    errText = ""

    oia.WaitForInputReady TIMEOUT_MS

    For i = 0 To UBound(STEPS)
        p = Split(STEPS(i), "|")
        cmd = UCase(Trim(p(0)))

        Select Case cmd
            Case "READ"
                AddPair p(1), Trim(ps.GetText(CInt(p(2)), CInt(p(3)), CInt(p(4))))
            Case "LABEL"
                AddPair p(1), GetAfterLabel(p(2), CInt(p(3)))
            Case "PGDN":  If Not PressKey("[pagedn]") Then Exit For
            Case "PGUP":  If Not PressKey("[pageup]") Then Exit For
            Case "ENTER": If Not PressKey("[enter]")  Then Exit For
            Case "F2":    If Not PressKey("[pf2]")    Then Exit For
            Case "F3":    If Not PressKey("[pf3]")    Then Exit For
            Case "F12":   If Not PressKey("[pf12]")   Then Exit For
            Case "KEY":   If Not PressKey(p(1))       Then Exit For
            Case "WAIT"
                If Not ps.WaitForString(p(1), 0, 0, TIMEOUT_MS, True, False) Then
                    errText = "Не дождались текста: " & p(1)
                    Exit For
                End If
            Case "PAUSE"
                ps.Wait CLng(p(1))
            Case Else
                errText = "Неизвестная команда: " & STEPS(i)
                Exit For
        End Select
    Next

    AddPair "Ошибка", errText
    WriteToFile

    If SHOW_MSGBOX Then
        If errText = "" Then
            MsgBox "Документ записан в файл", vbInformation, "Equation"
        Else
            MsgBox "Записано с ошибкой:" & vbCrLf & errText, vbExclamation, "Equation"
        End If
    End If
End Sub

' ---------- Нажатие клавиши с ожиданием смены экрана ----------
Function PressKey(k)
    Dim before, t
    PressKey = False
    before = ScreenText()

    ps.SendKeys k
    If Not oia.WaitForInputReady(TIMEOUT_MS) Then
        errText = "Таймаут после " & k
        Exit Function
    End If

    ' ждём, пока экран реально обновится (до 3 сек)
    t = 0
    Do While ScreenText() = before And t < 3000
        ps.Wait 100
        t = t + 100
    Loop

    If ScreenText() = before Then
        ' экран не сменился: последняя страница или клавиша недоступна
        errText = "Экран не изменился после " & k & ". " & Trim(ps.GetText(MSG_ROW, 1, ps.NumCols))
        Exit Function
    End If

    PressKey = True
End Function

Function ScreenText()
    ScreenText = ps.GetText(1, 1, ps.NumRows * ps.NumCols)
End Function

' ---------- Поиск значения по подписи ----------
Function GetAfterLabel(label, length)
    Dim r, c, rowText
    GetAfterLabel = ""
    For r = 1 To ps.NumRows
        rowText = ps.GetText(r, 1, ps.NumCols)
        c = InStr(1, rowText, label, vbTextCompare)
        If c > 0 Then
            c = c + Len(label)
            If c + length - 1 > ps.NumCols Then length = ps.NumCols - c + 1
            GetAfterLabel = Trim(ps.GetText(r, c, length))
            Exit Function
        End If
    Next
End Function

' ---------- Запись в файл (дописывание) ----------
Sub WriteToFile()
    Dim fso, f, fileName, isNew
    Set fso = CreateObject("Scripting.FileSystemObject")
    If Not fso.FolderExists(OUT_DIR) Then fso.CreateFolder OUT_DIR

    fileName = OUT_DIR & "eq_docs_" & Year(Date) & Right("0" & Month(Date), 2) & Right("0" & Day(Date), 2) & ".csv"
    isNew = Not fso.FileExists(fileName)

    If UNICODE_FILE Then
        Set f = fso.OpenTextFile(fileName, 8, True, -1)
    Else
        Set f = fso.OpenTextFile(fileName, 8, True, 0)
    End If
    If isNew Then f.WriteLine JoinCsv(headers)
    f.WriteLine JoinCsv(values)
    f.Close
End Sub

' ---------- Вспомогательные ----------
Sub AddPair(h, v)
    ReDim Preserve headers(UBound(headers) + 1)
    headers(UBound(headers)) = h
    ReDim Preserve values(UBound(values) + 1)
    values(UBound(values)) = v
End Sub

Function JoinCsv(arr)
    Dim i, s, v
    s = ""
    For i = 0 To UBound(arr)
        v = CStr(arr(i))
        If InStr(v, SEP) > 0 Or InStr(v, """") > 0 Then
            v = """" & Replace(v, """", """""") & """"
        End If
        If i > 0 Then s = s & SEP
        s = s & v
    Next
    JoinCsv = s
End Function
