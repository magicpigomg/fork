[PCOMM SCRIPT HEADER]
LANGUAGE=VBSCRIPT
DESCRIPTION=Equation: сбор полей документа по страницам в Excel
[PCOMM SCRIPT SOURCE]
OPTION EXPLICIT
autECLSession.SetConnectionByName(ThisSessionName)

' ================== НАСТРОЙКИ ==================
Const OUT_DIR     = "C:\EQ\"
Const MSG_ROW     = 24           ' строка сообщений Equation
Const TIMEOUT_MS  = 10000
Const SHOW_MSGBOX = True

' Названия полей из сценария, которые подставляются в "Результат"
Const F_NUM      = "Номер документа"
Const F_DOC_DATE = "Дата документа"
Const F_PUT_DATE = "Дата помещения"
Const F_SUM      = "Сумма"
Const FORMAT_PUT_DATE = True     ' True - "Дата помещения" тоже форматируется точками

' ================== СЦЕНАРИЙ ==================
'   READ|Название|строка|колонка|длина   - прочитать поле по координатам
'   LABEL|Название|Текст подписи|длина   - прочитать значение после подписи
'   PGDN / PGUP / ENTER / F2 / F3 / F12  - нажать клавишу
'   KEY|[pf5]                            - любая клавиша в формате PCOMM
'   WAIT|Текст                           - ждать появления текста на экране
'   PAUSE|мс                             - пауза
Dim STEPS
STEPS = Array( _
    "READ|Номер документа|4|20|16", _
    "READ|Дата документа|5|20|8", _
    "READ|Сумма|10|20|18", _
    "PGDN", _
    "READ|Счёт дебета|7|20|20", _
    "READ|Счёт кредита|8|20|20", _
    "PGDN", _
    "READ|Дата помещения|6|20|8", _
    "LABEL|Назначение|Назначение:|60", _
    "F2", _
    "READ|Доп. реквизит|6|25|30", _
    "F12" _
)
' ===============================================

Dim ps, oia, headers, values, errText
Set ps  = autECLSession.autECLPS
Set oia = autECLSession.autECLOIA

subMain

Sub subMain()
    Dim i, p, cmd
    headers = Array("Дата выгрузки")
    values  = Array(CStr(Now()))
    errText = ""

    oia.WaitForInputReady TIMEOUT_MS

    For i = 0 To UBound(STEPS)
        p = Split(STEPS(i), "|")
        cmd = UCase(Trim(p(0)))

        Select Case cmd
            Case "READ"
                AddPair p(1), Trim(ps.GetText(CInt(p(2)), CInt(p(3)), CInt(p(4))))
            Case "LABEL"
                AddPair p(1), GetAfterLabel(p(2), CInt(p(3)))
            Case "PGDN":  If Not PressKey("[pagedn]") Then Exit For
            Case "PGUP":  If Not PressKey("[pageup]") Then Exit For
            Case "ENTER": If Not PressKey("[enter]")  Then Exit For
            Case "F2":    If Not PressKey("[pf2]")    Then Exit For
            Case "F3":    If Not PressKey("[pf3]")    Then Exit For
            Case "F12":   If Not PressKey("[pf12]")   Then Exit For
            Case "KEY":   If Not PressKey(p(1))       Then Exit For
            Case "WAIT"
                If Not ps.WaitForString(p(1), 0, 0, TIMEOUT_MS, True, False) Then
                    errText = "Не дождались текста: " & p(1)
                    Exit For
                End If
            Case "PAUSE"
                ps.Wait CLng(p(1))
            Case Else
                errText = "Неизвестная команда: " & STEPS(i)
                Exit For
        End Select
    Next

    AddPair "Результат", BuildResult()
    AddPair "Ошибка", errText

    WriteToExcel

    If SHOW_MSGBOX Then
        If errText = "" Then
            MsgBox "Документ записан в файл", vbInformation, "Equation"
        Else
            MsgBox "Записано с ошибкой:" & vbCrLf & errText, vbExclamation, "Equation"
        End If
    End If
End Sub

' ---------- Колонка "Результат" ----------
Function BuildResult()
    Dim putDate
    putDate = GetValue(F_PUT_DATE)
    If FORMAT_PUT_DATE Then putDate = FormatEqDate(putDate)

    BuildResult = "ПТ № " & GetValue(F_NUM) & _
                  " от " & FormatEqDate(GetValue(F_DOC_DATE)) & _
                  " направлено в АИС ИДО " & putDate & _
                  " в сумме " & FormatEqSum(GetValue(F_SUM))
End Function

' "231026"            -> "23.10.26"
' "23102026"          -> "23.10.2026"
' "231026 10:39:39"   -> "23.10.26 10:39:39"
' "23102610:39:39"    -> "23.10.26 10:39:39"
Function FormatEqDate(s)
    Dim d, t, p
    s = Trim(s)
    d = s
    t = ""

    ' отделяем время: две цифры часа перед первым ":" и всё, что дальше
    p = InStr(s, ":")
    If p > 2 Then
        t = Trim(Mid(s, p - 2))
        d = Trim(Left(s, p - 3))
    End If

    ' форматируем только дату
    d = Replace(d, " ", "")
    If InStr(d, ".") = 0 And Len(d) >= 5 Then
        d = Left(d, 2) & "." & Mid(d, 3, 2) & "." & Mid(d, 5)
    End If

    If t <> "" Then
        FormatEqDate = d & " " & t
    Else
        FormatEqDate = d
    End If
End Function

Function GetValue(name)
    Dim i
    GetValue = ""
    For i = 0 To UBound(headers)
        If headers(i) = name Then GetValue = values(i): Exit Function
    Next
End Function

' ---------- Нажатие клавиши с ожиданием смены экрана ----------
Function PressKey(k)
    Dim before, t
    PressKey = False
    before = ScreenText()

    ps.SendKeys k
    If Not oia.WaitForInputReady(TIMEOUT_MS) Then
        errText = "Таймаут после " & k
        Exit Function
    End If

    t = 0
    Do While ScreenText() = before And t < 3000
        ps.Wait 100
        t = t + 100
    Loop

    If ScreenText() = before Then
        errText = "Экран не изменился после " & k & ". " & Trim(ps.GetText(MSG_ROW, 1, ps.NumCols))
        Exit Function
    End If
    PressKey = True
End Function

Function ScreenText()
    ScreenText = ps.GetText(1, 1, ps.NumRows * ps.NumCols)
End Function

' ---------- Поиск значения по подписи ----------
Function GetAfterLabel(label, length)
    Dim r, c, rowText
    GetAfterLabel = ""
    For r = 1 To ps.NumRows
        rowText = ps.GetText(r, 1, ps.NumCols)
        c = InStr(1, rowText, label, vbTextCompare)
        If c > 0 Then
            c = c + Len(label)
            If c + length - 1 > ps.NumCols Then length = ps.NumCols - c + 1
            GetAfterLabel = Trim(ps.GetText(r, c, length))
            Exit Function
        End If
    Next
End Function

' ---------- Запись в Excel (дописывание + автоширина) ----------
Sub WriteToExcel()
    Const xlUp = -4162, xlToLeft = -4159, xlOpenXMLWorkbook = 51
    Dim fso, fileName, xl, wb, ws, w, own, isNew, r, i, col, lastCol

    Set fso = CreateObject("Scripting.FileSystemObject")
    If Not fso.FolderExists(OUT_DIR) Then fso.CreateFolder OUT_DIR
    fileName = OUT_DIR & "eq_docs_" & Year(Date) & Right("0" & Month(Date), 2) & Right("0" & Day(Date), 2) & ".xlsx"

    On Error Resume Next

    ' 1. Если файл уже открыт у пользователя в Excel, пишем прямо в него
    Set wb = Nothing
    Set xl = GetObject(, "Excel.Application")
    If Err.Number = 0 Then
        For Each w In xl.Workbooks
            If LCase(w.FullName) = LCase(fileName) Then Set wb = w
        Next
    End If
    Err.Clear

    ' 2. Иначе открываем свой скрытый экземпляр Excel
    own = False
    If wb Is Nothing Then
        Set xl = CreateObject("Excel.Application")
        If Err.Number <> 0 Then errText = errText & " | Excel не запускается: " & Err.Description: Exit Sub
        own = True
        xl.Visible = False
        xl.DisplayAlerts = False
        isNew = Not fso.FileExists(fileName)
        If isNew Then
            Set wb = xl.Workbooks.Add
        Else
            Set wb = xl.Workbooks.Open(fileName)
        End If
    End If
    Set ws = wb.Worksheets(1)

    ' последняя колонка заголовка и первая свободная строка
    If ws.Cells(1, 1).Value = "" Then
        lastCol = 0
        r = 2
    Else
        lastCol = ws.Cells(1, ws.Columns.Count).End(xlToLeft).Column
        r = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row + 1
    End If

    ' пишем значения в колонку с таким же заголовком (новые колонки добавляются справа)
    For i = 0 To UBound(headers)
        col = 0
        Dim c
        For c = 1 To lastCol
            If ws.Cells(1, c).Value = headers(i) Then col = c: Exit For
        Next
        If col = 0 Then
            lastCol = lastCol + 1
            col = lastCol
            ws.Cells(1, col).Value = headers(i)
            ws.Cells(1, col).Font.Bold = True
        End If
        ws.Cells(r, col).NumberFormat = "@"       ' текст: не терять нули и длинные номера
        ws.Cells(r, col).Value = values(i)
    Next

    ws.Columns.AutoFit                            ' ширина под содержимое

    If own Then
        If isNew Then
            wb.SaveAs fileName, xlOpenXMLWorkbook
        Else
            wb.Save
        End If
        wb.Close False
        xl.Quit
    Else
        wb.Save
    End If

    If Err.Number <> 0 Then
        errText = errText & " | Ошибка Excel: " & Err.Description
        If own Then wb.Close False: xl.Quit
    End If
    On Error GoTo 0
End Sub

' ---------- Вспомогательные ----------
Sub AddPair(h, v)
    ReDim Preserve headers(UBound(headers) + 1)
    headers(UBound(headers)) = h
    ReDim Preserve values(UBound(values) + 1)
    values(UBound(values)) = v
End Sub
