[PCOMM SCRIPT HEADER]
LANGUAGE=VBSCRIPT
DESCRIPTION=Фильтр по сценарию
[PCOMM SCRIPT SOURCE]
OPTION EXPLICIT
autECLSession.SetConnectionByName(ThisSessionName)

' ================== СЦЕНАРИИ ==================
' Первый элемент - название в меню, дальше шаги:
'   TYPE|текст          напечатать текст с позиции курсора ({CLIP} - значение из буфера)
'   ENTER               Enter
'   KEY|[pf18]          любая клавиша PCOMM (Shift+F6 = [pf18])
'   CURSOR|строка|кол   поставить курсор
Dim SCENARIOS
SCENARIOS = Array( _
    Array("0104", _
        "TYPE|YU3", _
        "ENTER", _
        "TYPE|0104", _
        "ENTER", _
        "KEY|[pf18]", _
        "CURSOR|21|29", _
        "TYPE|=={CLIP}") _
)
' ===============================================

Dim ps, oia
Set ps  = autECLSession.autECLPS
Set oia = autECLSession.autECLOIA

subMain

Sub subMain()
    Dim menu, i, choice, sc, p, clip, txt

    ' --- меню ---
    menu = "Выберите сценарий:" & vbCrLf & vbCrLf
    For i = 0 To UBound(SCENARIOS)
        menu = menu & "  " & (i + 1) & " - " & SCENARIOS(i)(0) & vbCrLf
    Next
    menu = menu & vbCrLf & "Введите номер:"

    choice = InputBox(menu, "Фильтр Equation", "1")
    If choice = "" Then Exit Sub
    If Not IsNumeric(choice) Then Exit Sub
    choice = CInt(choice) - 1
    If choice < 0 Or choice > UBound(SCENARIOS) Then Exit Sub
    sc = SCENARIOS(choice)

    ' --- буфер обмена (только если сценарий его использует) ---
    clip = ""
    If InStr(Join(sc, " "), "{CLIP}") > 0 Then
        clip = GetClipboard()
        clip = Trim(Replace(Replace(clip, vbCr, ""), vbLf, ""))
        If clip = "" Then
            MsgBox "Буфер обмена пуст. Скопируйте значение и повторите.", vbExclamation, "Фильтр Equation"
            Exit Sub
        End If
    End If

    ' --- шаги ---
    For i = 1 To UBound(sc)
        p = Split(sc(i) & "|||", "|")
        Select Case UCase(Trim(p(0)))
            Case "TYPE"
                txt = Replace(p(1), "{CLIP}", clip)
                ps.SendKeys Replace(Replace(txt, "[", "[["), "]", "]]")
                ps.Wait 50
            Case "ENTER"
                If Not PressKey("[enter]") Then Exit Sub
            Case "KEY"
                If Not PressKey(p(1)) Then Exit Sub
            Case "CURSOR"
                ps.SetCursorPos CInt(p(1)), CInt(p(2))
        End Select
    Next
End Sub

' Нажать клавишу и дождаться ответа Equation
Function PressKey(k)
    Dim before, t
    before = ps.GetText(1, 1, ps.NumRows * ps.NumCols)
    ps.SendKeys k
    t = 0
    Do While ps.GetText(1, 1, ps.NumRows * ps.NumCols) = before And t < 2000
        ps.Wait 50
        t = t + 50
    Loop
    PressKey = oia.WaitForInputReady(15000)
    If Not PressKey Then MsgBox "Equation не ответил после " & k, vbExclamation, "Фильтр Equation"
End Function

' Чтение буфера обмена
Function GetClipboard()
    Dim s, sh, fso, tmp, st
    s = ""
    On Error Resume Next
    s = CreateObject("htmlfile").parentWindow.clipboardData.getData("Text")
    If Err.Number <> 0 Then s = ""
    If IsNull(s) Then s = ""
    Err.Clear
    On Error GoTo 0

    If s = "" Then
        Set sh = CreateObject("WScript.Shell")
        Set fso = CreateObject("Scripting.FileSystemObject")
        tmp = fso.GetSpecialFolder(2) & "\eq_clip.txt"
        sh.Run "powershell -NoProfile -WindowStyle Hidden -Command ""Get-Clipboard -Raw | Out-File -FilePath '" & tmp & "' -Encoding utf8""", 0, True
        If fso.FileExists(tmp) Then
            Set st = CreateObject("ADODB.Stream")
            st.Type = 2: st.Charset = "utf-8": st.Open
            st.LoadFromFile tmp
            s = st.ReadText
            st.Close
            fso.DeleteFile tmp
        End If
    End If
    GetClipboard = s
End Function
